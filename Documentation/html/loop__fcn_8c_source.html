18/* Automatic writing sensor data to serial */
19void autoread_fcn(uint32_t reg)
20{
21 if (_read_BV(reg, AUTOREAD_BIT))
22 {
23 /* Previous event time */
24 static uint32_t ticks = 0;
25
26 if (HAL_GetTick() > (ticks + set_data.time_a * 100))
27 {
28 ticks = HAL_GetTick();

51/* Function for handling valves */
52void valve_fcn(uint32_t * p_reg, float * p_set_pos)
53{
54 static uint32_t ticks = 0;
55
56 /* Default measured_data.pos is 0 0 0 */
57 static float stable_pos = 0;
58 static float required_pos = 0;
59
60 if (_read_BV(*p_reg, RUN_BIT))
61 {
62 if (_read_BV(*p_reg, PULSE_BIT) || _read_BV(*p_reg, TURN_BIT) || _read_BV(*p_reg, HOME_BIT))
63 {
64 /* Run is handled */
65 _clr_BV(*p_reg, RUN_BIT);
66
67 /* Duration of pulse */
68 ticks = HAL_GetTick();
69 /* Time is interpreted as tens of milliseconds */
70 ticks += (_read_BV(*p_reg, DIR_BIT)) ? ((_read_time_r(*p_reg)) * PULSE_REPRE) : ((_read_time_l(*p_reg)) * PULSE_REPRE);
71
72 /* 1 - right direction, 0 - left direction */
73 HAL_GPIO_WritePin(VALVE_R_GPIO_Port, VALVE_R_Pin, _read_BV(*p_reg, DIR_BIT));
74 HAL_GPIO_WritePin(VALVE_L_GPIO_Port, VALVE_L_Pin, !(_read_BV(*p_reg, DIR_BIT)));
75
76 /* Update stable position */
77 stable_pos = measured_data.pos[2];
78 /* For regulator */
79 *p_set_pos = stable_pos;
80
81 printf("stable_pos: %d\n", (int16_t)(stable_pos));
82 }
83 else
84 {
85 /* Satellite is moving, waiting for required position */
86 if ((fabsf(required_pos - measured_data.pos[2]) < ANGLE_TOL))
87 {
88 /* Create opposite pulse */
89 _tog_BV(*p_reg, DIR_BIT);
90 _set_BV(*p_reg, PULSE_BIT);
91
92 /* Redefine time according to angle speed */
93 _clr_time(*p_reg);
94 uint8_t time[2];
95 /* Positive velocity => right rotation => left valve for stop */
96 time[0] = !(_read_BV(*p_reg, DIR_BIT)) ? (uint8_t)(STOP_CONST_LEFT * fabsf(measured_data.gyr[2])) : set_data.time_l;
97 time[1] = (_read_BV(*p_reg, DIR_BIT)) ? (uint8_t)(STOP_CONST_RIGHT * measured_data.gyr[2]) : set_data.time_r;
98 _set_time(*p_reg, time[0], time[1]);
99
100 // printf("gyroscope: %d\n", _float2int(measured_data.gyr[2]));
101 // printf("break time: %d %d\n", time[0], time[1]);
102 }
103 }
104 }
105 else if ((HAL_GetTick() > ticks))
106 {
107 if (_read_BV(*p_reg, PULSE_BIT))
108 {
109 _clr_BV(*p_reg, PULSE_BIT);
110
111 valve_close();
112 }
113 else if (_read_BV(*p_reg, TURN_BIT))
114 {
115 _clr_BV(*p_reg, TURN_BIT);
116 /* When run is set, and pulse and turn bit clear, satellite is moving. */
117 /* Waiting for stopping pulse */
118 _set_BV(*p_reg, RUN_BIT);
119
120 valve_close();
121
122 /* Set duration of fly */
123 required_pos = stable_pos;
124 required_pos += (_read_BV(*p_reg, DIR_BIT)) ? (-set_data.angle) : set_data.angle;
125 sensor_deg_limit(&required_pos);
126
127 // printf("required_pos: %d\n", (int16_t)(required_pos));
128
129 }
130 else if (_read_BV(*p_reg, HOME_BIT))
131 {
132 _clr_BV(*p_reg, HOME_BIT);
133 _set_BV(*p_reg, RUN_BIT);
134
135 valve_close();
136
137 required_pos = (float)set_data.home;
138
139 // printf("required_pos: %d\n", (int16_t)(required_pos));
140 }
141 /* For regulation, it helps transfer regulation to home setting */
142 else if (_read_BV(*p_reg, RETURN_BIT))
143 {
144 /* Now I am flying */
145 _clr_BV(*p_reg, RETURN_BIT);
146 _set_BV(*p_reg, RUN_BIT);
147
148 required_pos = *p_set_pos;
149
150 // printf("required_pos: %d\n", (int16_t)(required_pos));
151 }
152 }
153}

155/* Regulation when cmd is not received */
156void regul_fcn(uint32_t * p_reg, float set_pos)
157{
158 static uint32_t ticks = 0;
159
160 /* No motion is processed */
161 if (!(_read_valve(*p_reg)))
162 {
163 /* Condition of running */
164 if ((_read_BV(*p_reg, CMD0_BIT)))
165 {
166 if (fabsf(measured_data.gyr[2]) > GYR_TOL)
167 {
168 regul_stop(p_reg, &ticks);
169 }
170 /* Cube is stable but position is different from required */
171 else if (fabsf(measured_data.pos[2] - set_pos) > ANGLE_TOL)
172 {
173 regul_return(p_reg, &ticks, set_pos);
174 }
175 }
176 }
177 else if (_read_BV(*p_reg, REG_BIT))
178 {
179 /* Condition of stopping */
180 if (HAL_GetTick() > ticks)
181 {
182 _clr_BV(*p_reg, REG_BIT);
183 valve_close();
184 // printf("Valve was closed\n");
185 }
186 }
187}

197/* Create stop pulse when position is forced to change */
198static void regul_stop(uint32_t * p_reg, uint32_t * p_ticks)
199{
200 _set_BV(*p_reg, REG_BIT);
201
202 if (measured_data.gyr[2] > 0)
203 {
204 HAL_GPIO_WritePin(VALVE_R_GPIO_Port, VALVE_R_Pin, 1);
205 *p_ticks = (uint32_t)(STOP_CONST_RIGHT * measured_data.gyr[2]);
206 }
207 else
208 {
209 HAL_GPIO_WritePin(VALVE_L_GPIO_Port, VALVE_L_Pin, 1);
210 /* Gyroscope data are negative!!! */
211 *p_ticks = (uint32_t)(-1 * STOP_CONST_LEFT * measured_data.gyr[2]);
212 }
213
214 /* Ignore too short pulses */
215 if (*p_ticks > PULSE_IGNORE)
216 {
217 *p_ticks += HAL_GetTick();
218 }
219 else
220 {
221 /* Too short */
222 _clr_BV(*p_reg, REG_BIT);
223 }
224}

226/* First pulse which start return to original position */
227static void regul_return(uint32_t * p_reg, uint32_t * p_ticks, float set_pos)
228{
229 /* Helps to regulate required destination */
230 _set_BV(*p_reg, REG_BIT);
231 _set_BV(*p_reg, RETURN_BIT);
232
233 *p_ticks = HAL_GetTick();
234
235 /* Choose direction */
236 if ((((measured_data.pos[2] - set_pos) > 0) && ((measured_data.pos[2] - set_pos) < 180)) || ((measured_data.pos[2] - set_pos) < -180))
237 {
238 _set_BV(*p_reg, DIR_BIT);
239
240 /* Interpreted as tens of milliseconds */
241 *p_ticks += set_data.time_r * PULSE_REPRE; //
242 HAL_GPIO_WritePin(VALVE_R_GPIO_Port, VALVE_R_Pin, 1);
243 // printf("Right valve was open\n");
244 }
245 else
246 {
247 _clr_BV(*p_reg, DIR_BIT);
248
249 /* Interpreted as tens of milliseconds */
250 *p_ticks += set_data.time_l * PULSE_REPRE;
251 HAL_GPIO_WritePin(VALVE_L_GPIO_Port, VALVE_L_Pin, 1);
252 // printf("Left valve was open\n");
253 }
