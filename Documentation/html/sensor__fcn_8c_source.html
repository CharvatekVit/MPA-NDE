47{
48 static uint32_t tick = 0;
49
50 if (HAL_GetTick() > tick)

60 for (uint8_t i = 0; i < 3; i++)

62 measured_data.acc[i] = acc[i];
63 measured_data.gyr[i] = gyr[i] - gyr_offset[i];
64 measured_data.mag[i] = mag[i];
65 measured_data.pos[i] += (measured_data.gyr[i] * MEAS_TIME * CAL_MUL) / 1000;

81/* Gyroscope calibration */
82static void sensor_calibration(float * gyr_offset)
83{
84 /* Temporary variables for calibration */
85 float acc[3];
86 float gyr[3];
87 float mag[3];
88
89 float offset[3] = {0};
90 uint8_t i;
91
92
93 for (i = 0; i < CAL_CYC_NUM; i++)
94 {
95 if (mpu9250_basic_read(acc, gyr, mag) == 0)
96 {
97 for (uint8_t j = 0; j < 3; j++)
98 {
99 offset[j] += gyr[j];
100 }
101 }
102
103 HAL_Delay(CAL_TIME);
104 }
105
106 for (uint8_t j = 0; j < 3; j++)
107 {
108 gyr_offset[j] = (offset[j] / i);
109 }
110}
